#!/usr/bin/env python3
"""Source-review disposition admission for merge-ready qualification (#8157).

Human prose, quoted examples, and historical observations are not disposition
fields. A current positive attestation is either:

- a column-zero readiness marker on the first substantive review line; or
- an explicit ``disposition: SOURCE_REVIEW_READY`` / ``disposition: REVIEW_READY``
  field on its own line outside fenced examples.

Native APPROVED / CHANGES_REQUESTED / DISMISSED reviews remain a separate path.

PR-body carrier (#4984): the same attestation may instead be one column-zero line in
the pull request description, ``Source review: REVIEW_READY <40-hex head>`` (or
``SOURCE_REVIEW_READY``). The full SHA must equal the candidate head, so a push makes
the line stale by construction. It is read only for a same-repository pull request
whose author association is trusted, which is the population that could already
self-attest with a COMMENT review. It adds a carrier, not a new authority: a native
CHANGES_REQUESTED or a trusted reviewer's current negative review still refuses.
The reason is operational: creating a review is throttled per account ("was submitted
too quickly"), which left reviewed candidates unable to request qualification.

Documented decision (#12814): free prose cannot be enumerated, so the marker line
is an allowlist built from the shapes of real review-of-record tails (full bodies
are fixtures under tools/repo-policy/tests/fixtures/review_bodies). An unfamiliar
tail refuses; a ready review refused for an unusual tail is the safe failure, and
the reviewer re-posts with the plain marker. Fenced blocks cannot hide a
retraction: retraction or blocker text inside a fence refuses. Later body prose is
screened by a best-effort negation list; the closed controls are the marker-line
allowlist and a later native review superseding the comment.
"""

READINESS_MARKERS = ("SOURCE_REVIEW_READY", "REVIEW_READY")
# A marker line: the marker alone, or the marker followed by a dash, comma or "for" and
# a tail. The tail is admitted only from a closed vocabulary (see tail_allowed): any word
# outside it refuses, so hedges, conditions and retractions cannot be enumerated away.
POSITIVE_LINE = re.compile(
    r"^ {0,3}(?:disposition:[ \t]*)?(?:SOURCE_REVIEW_READY|REVIEW_READY)"
    r"(?P<tail>(?:[ \t]*[,\u2014\u2013]|[ \t]+-[ \t]|[ \t]+for\b)[^\n]*)?[ \t]*$",
    re.IGNORECASE,
)
# Vocabulary derived from the first lines and `disposition:` lines of the 11 real
# REVIEW_READY native reviews on merged PRs #12736..#12793, plus connectors. Words that
# can hedge or retract (not, but, if, only, pending, blocked, fix, wait, ...) are absent
# on purpose; "change" is absent too and appears only inside the two literal phrases
# stripped first. An unusual but genuine tail is refused: that is the safe failure.
TAIL_VOCABULARY = frozenset(
    """for the a an and or of to then with by current published diagnostics candidate selected
    installed ordinary final fresh normal landed tree expected exact hash layout head source
    native proof findings integration qualification review complete completed squash protected

)

    re.compile(r"\bhash-layout[ \t]+change\b", re.IGNORECASE),

# Any line that speaks about disposition or readiness state. After the first line,
# such a line is only allowed when it is itself an exact positive line.
STATE_TOKEN = re.compile(

NEGATION = re.compile(

def normalize_body(body: str) -> str:
    text = (body or "").replace("\r\n", "\n").replace("\r", "\n")
    text = html.unescape(text).replace("\u2019", "'").replace("\u2018", "'")
    text = re.sub(r"[^\S\n]", " ", text)  # nbsp and other unicode spaces
    return text.replace("**", "").replace("__", "")


def split_body(body: str) -> tuple[list[str], list[str]] | None:
    """Return (visible lines, fenced lines); HTML comments are dropped.

    Returns ``None`` when the body is structurally ambiguous (unclosed fence or
    unclosed HTML comment); callers must refuse.
    """
    text = HTML_COMMENT.sub("", normalize_body(body))
    if "<!--" in text:
        return None
    visible: list[str] = []
    fenced: list[str] = []
    fence: tuple[str, int] | None = None
    for line in text.split("\n"):
        if fence is not None:
            char, size = fence
            if re.fullmatch(rf"^ {{0,3}}{re.escape(char)}{{{size},}}[ \t]*$", line):
                fence = None
            else:
                fenced.append(line)
            continue
        match = FENCE_OPEN.match(line)
        if match:
            fence = (match.group(1)[0], len(match.group(1)))
            continue
        visible.append(line)
    if fence is not None:
        return None
    return visible, fenced


def tail_allowed(tail: str) -> bool:
    """Whether a marker-line tail uses only the closed vocabulary."""
    text = tail
    for literal in TAIL_LITERALS:
        text = literal.sub(" ", text)
    # ASCII charset gate: any other character (CJK, emoji, "?", "!") refuses. The
    # leading em/en dash is the only non-ASCII character a real tail uses.
    if not re.fullmatch(r"[A-Za-z0-9 \t,.;:\-`'()/_\u2014\u2013]*", text):
        return False
    previous = ""
    for word in re.findall(r"[A-Za-z0-9]+", text):
        lowered = word.lower()
        if re.fullmatch(r"[0-9a-f]{7,40}", lowered):
            if previous != "head":  # a commit sha, only as "head <sha>"
                return False
        elif lowered not in TAIL_VOCABULARY:
            return False
        previous = lowered
    return True


def positive_line(line: str) -> bool:
    """Whether ``line`` is a marker line whose tail stays inside the vocabulary."""
    match = POSITIVE_LINE.match(normalize_body(line))
    if match is None:
        return False
    tail = match.group("tail") or ""
    return not STATE_TOKEN.search(tail) and tail_allowed(tail)


def comment_body_ready(body: str) -> bool:
    """Whether a COMMENT review body carries a current positive source disposition.

    The first substantive line must be a marker line with an allowed tail. Any later
    line that mentions disposition or a readiness state must itself be a positive
    line; retraction or blocker text refuses on later lines and inside fenced blocks.
    """
    parts = split_body(body)
    if parts is None:
        return False
    visible, fenced = parts
    substantive = [line for line in visible if line.strip()]
    if not substantive or not positive_line(substantive[0]):
        return False
    if any(STATE_TOKEN.search(line) or NEGATION.search(line) for line in fenced):
        return False
    return all(
        positive_line(line)
        or not (STATE_MENTION.search(line) or NEGATION.search(line))
        for line in substantive[1:]
    )


def comment_body_negative(body: str) -> bool:
    """Whether a COMMENT body speaks about disposition without being exactly positive.

    Ordinary chatter with no state mention is neutral. Structurally ambiguous bodies
    count as negative so they can never silently drop a refusal.
    """
    if comment_body_ready(body):
        return False
    parts = split_body(body)
    if parts is None:
        return True
    visible, fenced = parts
    return any(STATE_MENTION.search(line) for line in visible) or any(
        STATE_TOKEN.search(line) or NEGATION.search(line) for line in fenced
    )


def body_attests_head(body: str, head: str) -> bool:
    """Whether a pull request description carries the exact-head attestation line.

    Exactly one filled ``Source review:`` line may exist outside fences and HTML
    comments, its value must be a marker, one space and the full lowercase head SHA,
    and no fenced ``Source review:`` line may exist to be mistaken for it.
    """
    if not re.fullmatch(r"[0-9a-f]{40}", head or ""):
        return False
    parts = split_body(body)
    if parts is None:
        return False
    visible, fenced = parts
    if any(BODY_FIELD.match(line.strip()) for line in fenced):
        return False
    if any(BODY_COMPETING_FIELD.match(line) for line in (*visible, *fenced)):
        return False
    values = []
    for line in visible:
        match = BODY_FIELD.match(line)
        if match is not None and match.group("value"):
            values.append(match.group("value"))
    if len(values) != 1:
        return False
    match = BODY_FIELD_VALUE.match(values[0])
    return match is not None and match.group("sha") == head


def pull_allows_source_attestation(pull: dict[str, Any] | None, *, head: str) -> bool:
    """Whether the pull request itself carries a trusted exact-head attestation."""
    if not isinstance(pull, dict):
        return False
    if str(pull.get("author_association") or "").upper() not in TRUSTED_ASSOCIATIONS:
        return False
    head_ref = pull.get("head") if isinstance(pull.get("head"), dict) else {}
    base_ref = pull.get("base") if isinstance(pull.get("base"), dict) else {}
    head_repo = str((head_ref.get("repo") or {}).get("full_name") or "")
    base_repo = str((base_ref.get("repo") or {}).get("full_name") or "")
    if not head_repo or head_repo != base_repo:
        return False
    # The live head must be the head being admitted: a description written for an
    # earlier head never attests a later one.
    if str(head_ref.get("sha") or "") != head:
        return False
    return body_attests_head(str(pull.get("body") or ""), head)


def _order(review: dict[str, Any]) -> tuple[str, int]:
    return (str(review.get("submitted_at") or ""), int(review.get("id") or 0))


def _trusted(review: dict[str, Any]) -> bool:
    return str(review.get("author_association") or "").upper() in TRUSTED_ASSOCIATIONS


def _latest_by_login(reviews: list[dict[str, Any]]) -> list[dict[str, Any]]:
    by_login: dict[str, dict[str, Any]] = {}
    for index, review in enumerate(reviews):
        login = str((review.get("user") or {}).get("login") or "")
        if not login:
            # A deleted account yields user: null. Keep each such review as its own
            # identity so a blocking review is never silently dropped.
            review_id = review.get("id")
            login = f"<anonymous:{review_id if review_id is not None else f'index-{index}'}>"
        current = by_login.get(login)
        if current is None or _order(review) >= _order(current):
            by_login[login] = review
    return list(by_login.values())


def _latest_decisive(reviews: list[dict[str, Any]]) -> list[dict[str, Any]]:
    decisive = [r for r in reviews if str(r.get("state") or "") in DECISIVE_STATES]
    return _latest_by_login(decisive)


def reviews_allow_source_attestation(
    reviews: list[dict[str, Any]],
    *,
    head: str,
    author: str | None = None,
    pull: dict[str, Any] | None = None,
) -> bool:
    """Whether review evidence admits qualification for ``head``."""
    del author  # same-account COMMENT attestation is allowed; trust comes from author_association.
    if any(
        str(r.get("state") or "") != "PENDING" and not r.get("submitted_at")
        for r in reviews
    ):
        return False  # cannot be ordered, so it cannot be proven superseded
    native = _latest_decisive(reviews)
    if any(str(r.get("state") or "") == "CHANGES_REQUESTED" for r in native):
        return False
    if any(
        str(r.get("commit_id") or "") == head
        and str(r.get("state") or "") == "APPROVED"
        and _trusted(r)
        for r in native
    ):
        return True
    commented = [
        r
        for r in _latest_by_login(reviews)
        if str(r.get("commit_id") or "") == head
        and _trusted(r)
        and str(r.get("state") or "") == "COMMENTED"
    ]
    # Any trusted reviewer's current negative disposition refuses, even when another
    # trusted reviewer's current comment is ready.
    if any(comment_body_negative(str(r.get("body") or "")) for r in commented):
        return False
    if any(comment_body_ready(str(r.get("body") or "")) for r in commented):
        return True
    # The description is only another carrier, so it must not be easier to satisfy
    # than a ready review from the same people. A thread reply creates a neutral
    # COMMENTED review that replaces its author's earlier one in the latest-per-login
    # view above; here each trusted reviewer's latest current-head statement that
    # says anything (ready or negative) decides, and a negative one refuses.
    stated = [
        r
        for r in reviews
        if str(r.get("commit_id") or "") == head
        and _trusted(r)
        and str(r.get("state") or "") == "COMMENTED"
        and (
            comment_body_ready(str(r.get("body") or ""))
            or comment_body_negative(str(r.get("body") or ""))
        )
    ]
    if any(comment_body_negative(str(r.get("body") or "")) for r in _latest_by_login(stated)):
        return False
    return pull_allows_source_attestation(pull, head=head)


def parse_paginated(text: str) -> list[dict[str, Any]]:
    """Concatenate the JSON arrays ``gh api --paginate`` prints, one per page."""
    decoder = json.JSONDecoder()
    reviews: list[dict[str, Any]] = []
    index = 0
    while index < len(text):
        if text[index].isspace():
            index += 1
            continue
        page, index = decoder.raw_decode(text, index)
        if not isinstance(page, list) or not all(isinstance(item, dict) for item in page):
            raise ValueError("review page is not a list of review objects")
        reviews.extend(page)
    return reviews


def fetch_reviews(repository: str, pr_number: str) -> list[dict[str, Any]]:
    """Read every native review page; any failure raises so callers fail closed."""
    out = subprocess.run(
        [
            "gh",
            "api",
            "--paginate",
            f"repos/{repository}/pulls/{pr_number}/reviews?per_page=100",
        ],
        check=True,
        capture_output=True,
        text=True,
        encoding="utf-8",
        timeout=120,
    ).stdout
    return parse_paginated(out)


def fetch_pull(repository: str, pr_number: str) -> dict[str, Any]:
    """Read the pull request object; any failure raises."""
    out = subprocess.run(
        ["gh", "api", f"repos/{repository}/pulls/{pr_number}"],
        check=True,
        capture_output=True,
        text=True,
        encoding="utf-8",
        timeout=120,
    ).stdout
    pull = json.loads(out)
    if not isinstance(pull, dict):
        raise ValueError("pull request response is not an object")
    return pull

def main() -> int:
    parser = argparse.ArgumentParser()
    parser.add_argument("--self-test", action="store_true")
    # FileType returns the existing sys.stdin for '-', ignoring its encoding
    # argument. Read bytes so piped GitHub JSON also has an explicit UTF-8 boundary.
    parser.add_argument("--check-reviews", type=argparse.FileType("rb"))
    parser.add_argument(
        "--check-pull",
        type=argparse.FileType("rb"),
        help="pull request object (JSON) to read the Source review line from, with --check-reviews",
    )
    parser.add_argument("--repository", help="owner/name; read reviews with gh instead of a file")
    parser.add_argument("--pr", help="Pull request number for --repository")
    parser.add_argument("--head", help="Exact candidate head SHA")
    parser.add_argument("--author", help="PR author login (reserved for future custody checks)")
    args = parser.parse_args()
    if args.self_test:
        self_test()
        return 0
    live = args.repository is not None and args.pr is not None
    if not args.head or (args.check_reviews is None and not live):
        parser.error(
            "--head plus --check-reviews or --repository/--pr are required unless --self-test is set"
        )
    if live:
        try:
            reviews = fetch_reviews(args.repository, args.pr)
        except (OSError, subprocess.SubprocessError, ValueError) as exc:
            print(
                f"::error::source review attestation refused: could not read reviews: {exc}",
                file=sys.stderr,
            )
            return 1
        try:
            pull = fetch_pull(args.repository, args.pr)
        except (OSError, subprocess.SubprocessError, ValueError) as exc:
            # The description is an optional carrier: without it only reviews count.
            print(
                f"::warning::could not read the pull request description ({exc}); "
                "only native reviews are considered for the source review attestation",
                file=sys.stderr,
            )
            pull = None
    else:
        try:
            reviews = parse_paginated(args.check_reviews.read().decode("utf-8"))
            pull = None
            if args.check_pull is not None:
                pull = json.loads(args.check_pull.read().decode("utf-8"))
                if not isinstance(pull, dict):
                    raise ValueError("pull request data is not an object")
        except ValueError as exc:
            print(
                f"::error::source review attestation refused: malformed review data: {exc}",
                file=sys.stderr,
            )
            return 1
    if not reviews_allow_source_attestation(
        reviews, head=args.head, author=args.author, pull=pull
    ):
        print(
            "::error::source review attestation refused: a trusted reviewer "
            "(OWNER/MEMBER/COLLABORATOR) must post a current-head COMMENT whose first line is "
            "exactly SOURCE_REVIEW_READY or REVIEW_READY (or disposition: of either) with no "
            "contradicting disposition text, or a native APPROVED on the head; narrative "
            "tokens and quoted examples do not count (#8157). When GitHub throttles review "
            "writes, a same-repository pull request may instead carry one description line "
            f"`Source review: REVIEW_READY {args.head}` with the full current head SHA (#4984)",
            file=sys.stderr,
        )
        return 1
    return 0

    raise SystemExit(main())

//! #8157: shared source-review disposition verifier used by merge-ready admission.
use std::io::Write;
use std::path::PathBuf;
use std::process::{Command, Output, Stdio};

const SCRIPT: &str = ".github/actions/pr-candidate-admission/review_disposition.py";
const REFUSED: &str = "source review attestation refused";

fn script() -> Result<PathBuf, String> {
    let manifest = std::path::Path::new(env!("CARGO_MANIFEST_DIR"));
    let root = manifest
        .parent()
        .and_then(|p| p.parent())
        .ok_or("repository root not found")?;
    Ok(root.join(SCRIPT))
}

fn check_reviews(reviews: &serde_json::Value, head: &str) -> Result<Output, String> {
    check_review_bytes(reviews.to_string().as_bytes(), head, false)
}

fn check_review_bytes(bytes: &[u8], head: &str, legacy_stdin: bool) -> Result<Output, String> {
    check_review_bytes_with_pull(bytes, head, legacy_stdin, None)
}

/// #4984: reviews on stdin plus the pull request object read from `pull`.
fn check_reviews_with_pull(
    reviews: &serde_json::Value,
    pull: &serde_json::Value,
    head: &str,
    tag: &str,
) -> Result<Output, String> {
    let path = std::env::temp_dir().join(format!(
        "review-disposition-pull-{}-{tag}.json",
        std::process::id()
    ));
    std::fs::write(&path, pull.to_string()).map_err(|e| format!("write pull: {e}"))?;
    let output =
        check_review_bytes_with_pull(reviews.to_string().as_bytes(), head, false, Some(&path));
    let _ = std::fs::remove_file(&path);
    output
}

fn check_review_bytes_with_pull(
    bytes: &[u8],
    head: &str,
    legacy_stdin: bool,
    pull: Option<&std::path::Path>,
) -> Result<Output, String> {
    let mut command = Command::new("python3");
    command.arg("-I");
    if legacy_stdin {
        // Force the Windows legacy decoding boundary even on UTF-8 hosts.
        command.args(["-c", "import runpy,sys; sys.stdin.reconfigure(encoding='cp1252'); path=sys.argv.pop(1); runpy.run_path(path, run_name='__main__')"]);
    }
    command
        .arg(script()?)
        .args(["--check-reviews", "-", "--head", head]);
    if let Some(pull) = pull {
        child_args_pull(&mut command, pull);
    }
    let mut child = command
        .stdin(Stdio::piped())
        .stdout(Stdio::piped())
        .stderr(Stdio::piped())
        .spawn()
        .map_err(|e| format!("spawn python3: {e}"))?;
    child
        .stdin
        .take()
        .ok_or("stdin not piped")?
        .write_all(bytes)
        .map_err(|e| format!("write reviews: {e}"))?;
    child
        .wait_with_output()
        .map_err(|e| format!("wait for verifier: {e}"))
}

/// Admit means exit 0; refuse means exit 1 with the refusal line and no crash.
fn expect_outcome(output: &Output, admit: bool, what: &str) -> Result<(), String> {
    let stderr = String::from_utf8_lossy(&output.stderr);
    let ok = if admit {
        output.status.code() == Some(0)
    } else {
        output.status.code() == Some(1) && stderr.contains(REFUSED) && !stderr.contains("Traceback")
    };
    if ok {
        Ok(())
    } else {
        Err(format!(
            "{what}: expected {} got {:?} stderr={stderr}",
            if admit { "admit" } else { "refusal" },
            output.status.code()
        ))
    }
}

    })
}

fn step_id(block: &str) -> Option<&str> {
    block
        .lines()
        .find_map(|line| line.strip_prefix("        id: "))
}

fn contract_violations(workflow: &str) -> Vec<String> {
    let blocks: Vec<_> = workflow.split("\n      - ").skip(1).collect();
    let mut findings = Vec::new();
    let mut previous = None;
    for &(id, command) in REQUIRED_GATES {
        let matches: Vec<_> = blocks
            .iter()
            .enumerate()
            .filter(|(_, block)| step_id(block) == Some(id))
            .collect();
        if matches.len() != 1 {
            findings.push(format!("{id}: expected one step"));
            continue;
        }
        let Some(&(index, block)) = matches.first() else {
            continue;
        };
        if !block.lines().any(|line| {
            let line = line.trim();
            line.strip_prefix("run: ").unwrap_or(line) == command
        }) {
            findings.push(format!("{id}: exact command missing"));
        }
        if block.lines().any(|line| {
            (line.starts_with("        if:")
                && !(id == "rust-tests" && line == SHARDED_TEST_CONDITION))
                || line.starts_with("        continue-on-error:")
        }) {
            findings.push(format!("{id}: required step must remain unconditional"));
        }
        let command_count = workflow
            .lines()
            .map(str::trim)
            .map(|line| line.strip_prefix("run: ").unwrap_or(line))
            .filter(|line| *line == command)
            .count();
        if command_count != 1 {
            findings.push(format!("{id}: expected one command invocation"));
        }
        if previous.is_some_and(|prior| prior >= index) {
            findings.push(format!("{id}: gate order changed"));
        }
        previous = Some(index);
    }
    for (name, id) in [
        ("Prepare independent user-text base", "user_text_base"),
        (
            "Recover pinned base compiler arity if needed",
            "user_text_recovery",
        ),
        (
            "User-text goldens judged by the merge-base matcher",
            "user_text_matcher",
        ),
    ] {
        let matches: Vec<_> = blocks
            .iter()
            .filter(|block| block.lines().next() == Some(&format!("name: {name}")))
            .collect();
        if matches.len() != 1 || step_id(matches[0]) != Some(id) {
            findings.push(format!("{name}: expected unique step with id {id}"));
        }
    }
    let summaries: Vec<_> = blocks
        .iter()
        .filter(|block| step_id(block) == Some("gate_summary"))
        .collect();
    if summaries.len() != 1 {
        findings.push("summary: expected one step".to_owned());
        return findings;
    }
    let Some(summary) = summaries.first() else {
        return findings;
    };
    if !summary.lines().any(|line| line == "        if: always()") {
        findings.push("summary: must run after non-success".to_owned());
    }
    for id in REQUIRED_GATES
        .iter()
        .map(|(id, _)| *id)
        .chain(std::iter::once("pr_evidence"))
    {
        let variable = format!("{}_RESULT", id.replace('-', "_").to_ascii_uppercase());
        let binding = format!("          {variable}: ${{{{ steps.{id}.outcome }}}}");
        // The two pr_evidence rows share one printf to preserve the workflow's
        // reviewed executable-line budget. Pin both argument positions so the
        // compact form cannot swap or fabricate the required outcome.
        let row = if id == "pr_evidence" {
            format!("            {COMBINED_PR_EVIDENCE_PRINTF}")
        } else {
            format!("            printf '| {id} | %s |\\n' \"${{{variable}:-not_reported}}\"")
        };
        if !summary.lines().any(|line| line == binding) || !summary.lines().any(|line| line == row)
        {
            findings.push(format!("summary: {id} must retain its actual outcome"));
        }
    }
    for id in ["user_text_base", "user_text_recovery", "user_text_matcher"] {
        let variable = format!("{}_RESULT", id.to_ascii_uppercase());
        let binding = format!("          {variable}: ${{{{ steps.{id}.outcome }}}}");
        if !summary.lines().any(|line| line == binding) {
            findings.push(format!("summary: {id} must bind its actual outcome"));
        }
    }
    if !summary.lines().any(|line| {
        line == "          PR_EVIDENCE_SCOPE: ${{ steps.pr_evidence.outputs.evidence }}"
    }) {
        findings.push("summary: pr_evidence scope must retain its actual output".to_owned());
    }
    findings
}

#[test]
fn rust_gate_workflow_preserves_complete_ordered_command_inventory() {
    let findings = contract_violations(WORKFLOW);
    assert!(findings.is_empty(), "{findings:?}");
}

#[test]
fn user_text_summary_rejects_missing_or_swapped_outcome_bindings() {
    for id in ["user_text_base", "user_text_recovery", "user_text_matcher"] {
        let binding = format!("steps.{id}.outcome");
        for replacement in ["success", "steps.precommit.outcome"] {
            let changed = WORKFLOW.replace(&binding, replacement);
            assert_ne!(changed, WORKFLOW);
            assert!(
                contract_violations(&changed)
                    .contains(&format!("summary: {id} must bind its actual outcome"))
            );
        }
        let step = format!("        id: {id}");
        let changed = WORKFLOW.replace(&step, "        id: wrong_step");
        assert_ne!(changed, WORKFLOW);
        assert!(
            contract_violations(&changed)
                .iter()
                .any(|finding| finding.contains(id))
        );
    }
}

    fn new(case: &str) -> std::io::Result<Self> {
        let path = std::env::temp_dir().join(format!(
            "ripr-user-text-summary-{}-{case}",
            std::process::id()
        ));
        std::fs::File::create_new(&path)?;
        Ok(Self(path))
    }
}

#[cfg(unix)]
impl Drop for SummaryScratch {
    fn drop(&mut self) {
        let _ = std::fs::remove_file(&self.0);
    }
}

#[cfg(unix)]
#[test]
fn user_text_summary_reports_each_actual_outcome_and_missing_value()
-> Result<(), Box<dyn std::error::Error>> {
    use std::fs;
    use std::process::Command;

    let summary = WORKFLOW
        .split("\n      - ")
        .find(|block| step_id(block) == Some("gate_summary"))
        .ok_or("gate summary missing")?;
    let script = summary
        .split_once("        run: |\n")
        .ok_or("summary shell missing")?
        .1
        .lines()
        .map(|line| line.strip_prefix("          ").unwrap_or(line))
        .collect::<Vec<_>>()
        .join("\n");
    let variables = [
        "USER_TEXT_BASE_RESULT",
        "USER_TEXT_RECOVERY_RESULT",
        "USER_TEXT_MATCHER_RESULT",
    ];
    let labels = ["user_text_base", "user_text_recovery", "user_text_matcher"];
    let scratch = SummaryScratch::new("outcomes")?;
    let path = &scratch.0;
    for target in 0..3 {
        for outcome in ["success", "failure", "skipped", "cancelled", ""] {
            let mut command = Command::new("bash");
            command
                .arg("-eo")
                .arg("pipefail")
                .arg("-c")
                .arg(&script)
                .env_clear()
                .env("GITHUB_STEP_SUMMARY", path)
                .env("GITHUB_SHA", "workflow-sha")
                .env("SUBJECT_HEAD", "subject-head")
                .env("INTEGRATION_SUBJECT", "integration-sha")
                .env("INTEGRATION_TREE", "integration-tree")
                .env("GITHUB_RUN_ID", "run-id")
                .env("GITHUB_RUN_ATTEMPT", "attempt-id")
                .env("EXECUTION_ROUTE", "route")
                .env("PRECOMMIT_RESULT", "skipped");
            for (index, variable) in variables.iter().enumerate() {
                if index != target || !outcome.is_empty() {
                    command.env(variable, if index == target { outcome } else { "success" });
                }
            }
            fs::write(path, "")?;
            assert!(command.status()?.success());
            let rendered = fs::read_to_string(path)?;
            for (index, label) in labels.iter().enumerate() {
                let expected = if index == target {
                    if outcome.is_empty() {
                        "not_reported"
                    } else {
                        outcome
                    }
                } else {
                    "success"
                };
                assert!(
                    rendered.contains(&format!("| {label} | {expected} |")),
                    "missing or wrong row: {rendered}"
                );
            }
            for expected in [
                "workflow-sha",
                "subject-head",
                "integration-sha",
                "integration-tree",
                "run-id",
                "attempt-id",
                "route",
                "| precommit | skipped |",
            ] {
                assert!(
                    rendered.contains(expected),
                    "missing identity or downstream result {expected}: {rendered}"
                );
            }
        }
    }
    fs::remove_file(path)?;
    Ok(())
}

#[cfg(unix)]
#[test]
fn user_text_summary_scratch_cleans_up_on_success_error_and_unwind()
-> Result<(), Box<dyn std::error::Error>> {
    let scratch = SummaryScratch::new("normal")?;
    let normal_path = scratch.0.clone();
    drop(scratch);
    assert!(!normal_path.exists());

    let error_path = std::env::temp_dir().join(format!(
        "ripr-user-text-summary-{}-error",
        std::process::id()
    ));
    let error = (|| -> std::io::Result<()> {
        let scratch = SummaryScratch::new("error")?;
        // Pin the exact scratch path so a naming-scheme change cannot let the
        // error and unwind arms inspect an unrelated path.
        assert_eq!(scratch.0, error_path);
        assert!(error_path.exists());
        Err(std::io::Error::other("injected error"))
    })();
    assert!(matches!(error, Err(ref error) if error.to_string() == "injected error"));
    assert!(!error_path.exists());

    let unwind_path = std::env::temp_dir().join(format!(
        "ripr-user-text-summary-{}-unwind",
        std::process::id()
    ));
    let unwind = std::panic::catch_unwind(|| -> std::io::Result<()> {
        let scratch = SummaryScratch::new("unwind")?;
        assert_eq!(scratch.0, unwind_path);
        assert!(unwind_path.exists());
        std::panic::resume_unwind(Box::new("injected unwind"));
    });
    // `assert!(x.is_err())` trips clippy::assertions_on_result_states and
    // `unwrap_err` trips clippy::unwrap_used; `matches!` keeps the intent and
    // pins the injected payload so an unrelated panic cannot pass as it.
    assert!(matches!(unwind, Err(ref payload)
        if payload.downcast_ref::<&str>() == Some(&"injected unwind")));
    assert!(!unwind_path.exists());
    Ok(())
}

#[test]
fn user_text_summary_scratch_preserves_replacement_directory()
-> Result<(), Box<dyn std::error::Error>> {
    use std::fs;

    let scratch = SummaryScratch::new("replacement")?;
    let replacement = scratch.0.clone();
    fs::remove_file(&replacement)?;
    fs::create_dir(&replacement)?;
    let sentinel = replacement.join("summary.md");
    fs::write(&sentinel, "must survive")?;

}

const FIXTURE_PREPARATION_NAME: &str = "name: Prepare locked owner-return fixture dependencies";
const FALLIBLE_FETCH: &str = "cargo fetch --locked --manifest-path fixtures/owner_return_pin_fallible_literal/input/Cargo.toml";
const JSON_FETCH: &str = "cargo fetch --locked --manifest-path fixtures/owner_return_pin_json_real_literal/input/Cargo.toml";
const PROPTEST_FETCH: &str = "cargo fetch --locked --manifest-path crates/ripr/tests/fixtures/proptest_singleton/input/Cargo.toml";

fn prepares_locked_owner_return_fixtures(workflow: &str) -> bool {
    let blocks: Vec<_> = workflow.split("\n      - ").skip(1).collect();
    let preparations: Vec<_> = blocks
        .iter()
        .enumerate()
        .filter(|(_, block)| block.lines().next() == Some(FIXTURE_PREPARATION_NAME))
        .collect();
    let [(index, block)] = preparations.as_slice() else {
        return false;
    };
    // Exact bare commands retain lock enforcement and failure propagation;
    // no condition, alternate Cargo home, or suppressed failure is accepted.
    let lines: Vec<_> = block
        .lines()
        .map(str::trim)
        .filter(|line| !line.is_empty())
        .collect();
    lines
        == [
            FIXTURE_PREPARATION_NAME,
            "run: |",
            FALLIBLE_FETCH,
            JSON_FETCH,
            PROPTEST_FETCH,
        ]
        && blocks[..*index]
            .iter()
            .any(|block| block.starts_with("uses: dtolnay/rust-toolchain@"))
        && ["precommit", "rust-tests"].iter().all(|id| {
            blocks[*index + 1..]
                .iter()
                .any(|block| step_id(block) == Some(id))
        })
        && [FALLIBLE_FETCH, JSON_FETCH, PROPTEST_FETCH]
            .iter()
            .all(|command| {
                workflow
                    .lines()
                    .filter(|line| line.trim() == *command)
                    .count()
                    == 1
            })
}

}

fn test_build_script(workflow: &str) -> Result<String, String> {
    let step = workflow
        .split_once("      - name: Build the test binaries one memory-heavy crate at a time\n")
        .ok_or_else(|| "missing serialized test-build step".to_owned())?
        .1
        .split("\n      - ")
        .next()
        .ok_or_else(|| "missing test-build step body".to_owned())?;
    let body = step
        .split_once("        run: |\n")
        .ok_or_else(|| "test-build step must keep a literal run block".to_owned())?
        .1;
    Ok(body
        .lines()
        .map(|line| line.strip_prefix("          ").unwrap_or(line))
        .collect::<Vec<_>>()
        .join("\n"))
}

fn checks_with_prepared_xtask(workflow: &str) -> bool {
    let commands = [
        "          build_phase workspace cargo test --no-run --locked --workspace",
        "        run: target/debug/xtask check-covered-by",
        "          strip --strip-debug -o target/xtask-binary/xtask target/debug/xtask",
        "      - name: Required Rust tests (shard ${{ matrix.shard }} of 4)",
    ];
    let mut previous = None;
    for command in commands {
        let positions: Vec<_> = workflow
            .lines()
            .enumerate()
            .filter_map(|(index, line)| (line == command).then_some(index))
            .collect();
        let [position] = positions.as_slice() else {
            return false;
        };
        if previous.is_some_and(|before| before >= *position) {
            return false;
        }
        previous = Some(*position);
    }
    let Some(check) = workflow.split("\n      - ").find(|block| {
        block.starts_with("name: Check covered-by with the prepared workspace binary\n")
    }) else {
        return false;
    };
    check
        .lines()
        .filter(|line| !line.trim().is_empty() && !line.trim_start().starts_with('#'))
        .collect::<Vec<_>>()
        .join("\n")
        == "name: Check covered-by with the prepared workspace binary\n        timeout-minutes: 20\n        run: target/debug/xtask check-covered-by"
        && !workflow.contains("cargo xtask check-covered-by")
}
