//! Membership pricing rules for the storefront.

/// Returns the discount percentage applied at checkout for a member age.
///
/// Members aged 65 and over receive the senior discount.
pub fn discount_percent(age: u32) -> u32 {
    if age > 65 {
        10
    } else {
        0
    }
}
