//! A Some(literal) early return observed through an as-binding pattern with the exact payload.

pub fn lookup(k: u32) -> Option<u32> {
    if k == 1 {
        return Some(3);
    }
    None
}

#[cfg(test)]
mod tests {
    use super::lookup;

    #[test]
    fn key_one_finds_three() {
        assert!(matches!(lookup(1), Some(x @ 3)));
    }
}
