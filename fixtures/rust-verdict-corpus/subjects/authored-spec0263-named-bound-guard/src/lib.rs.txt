/// The largest value left out of the list.
pub const LIMIT: u32 = 10;

/// Records `x` only when it is large.
pub fn add(v: &mut Vec<u32>, x: u32) {
    if x > LIMIT {
        v.push(x);
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn adds_nothing_for_small_values() {
        let mut v = Vec::new();
        add(&mut v, 4);
        assert_eq!(v, Vec::<u32>::new());
    }
}
