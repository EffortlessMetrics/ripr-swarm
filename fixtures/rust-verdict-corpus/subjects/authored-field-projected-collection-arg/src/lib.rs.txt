/// A list of recorded values.
pub struct Bag(pub Vec<u32>);

/// Records `x`.
pub fn add(v: &mut Vec<u32>, x: u32) {
    v.push(x);
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn records_into_the_bag() {
        let mut b = Bag(Vec::new());
        add(&mut b.0, 11);
        assert_eq!(b.0, vec![11]);
    }
}
