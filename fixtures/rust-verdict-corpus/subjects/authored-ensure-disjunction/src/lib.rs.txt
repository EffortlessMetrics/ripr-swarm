pub fn score(x: u32) -> u32 {
    x * 2
}

#[cfg(test)]
mod tests {
    use super::*;

    macro_rules! ensure {
        ($cond:expr) => {
            if !$cond {
                return Err(String::from("ensure failed"));
            }
        };
    }

    #[test]
    fn score_is_nonzero_or_pinned() -> Result<(), String> {
        ensure!(score(2) != 0 || score(3) == 6);
        Ok(())
    }
}
