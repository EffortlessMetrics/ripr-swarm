#[derive(Debug, PartialEq)]
pub enum E {
    Bad,
    Big,
}

pub fn check(x: u32) -> Result<u32, E> {
    if x > 10 {
        return Err(E::Big);
    }
    Ok(x)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn check_rejects_twenty() {
        assert!(matches!(check(20), Err(_e)));
    }
}
