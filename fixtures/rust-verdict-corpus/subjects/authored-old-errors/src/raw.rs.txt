//! An unsafe block that shares its line with the surrounding statement.

/// The first byte plus one, for a non-empty slice.
pub fn head(bytes: &[u8]) -> u32 {
    assert!(!bytes.is_empty());
    let first = unsafe { *bytes.get_unchecked(0) };
    u32::from(first) + 1
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn head_reads_the_first_byte() {
        assert_eq!(head(&[5, 9]), 6);
    }
}
