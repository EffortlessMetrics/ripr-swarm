//! A test that checks the owner only with `debug_assert_eq!` (#8051).

/// Shipping cost in cents for a parcel of `grams`.
pub fn shipping_cents(grams: u32) -> u32 {
    let base = 250;
    base + grams / 100 * 40
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn a_kilo_parcel_costs_six_fifty() {
        debug_assert_eq!(shipping_cents(1000), 650);
    }
}
