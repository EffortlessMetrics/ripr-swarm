//! The guard condition is a helper call, so its meaning is opaque.

pub fn tripled(x: u32) -> u32 {
    x * 3
}

#[cfg(test)]
mod tests {
    use super::*;

    fn looks_right(value: u32) -> bool {
        value > 0
    }

    #[test]
    fn two_triples_to_six_g_opaque() -> Result<(), String> {
        let value = tripled(2);
        if !looks_right(value) {
            return Err(format!("unexpected {value}"));
        }
        Ok(())
    }
}
