#[derive(Debug, Clone, PartialEq)]
pub struct Line {
    pub sku: String,
    pub qty: u32,
}

#[derive(Default)]
pub struct Order {
    pub lines: Vec<Line>,
}

impl Order {
    pub fn add_row(&mut self, sku: &str, qty: u32) {
        self.lines.push(Line { sku: sku.to_uppercase(), qty });
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn add_row_records_the_line_p_owner() {
        let mut order = Order::default();
        let mut expected = Order::default();
        expected.add_row("bolt", 2);
        order.add_row("bolt", 2);
        assert_eq!(order.lines, expected.lines);
    }
}
