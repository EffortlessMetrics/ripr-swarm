#[derive(Debug, PartialEq)]
pub enum Entry {
    Beat { n: u32 },
}

pub struct Pulse {
    pub log: Vec<Entry>,
    pub locked: bool,
}

impl Pulse {
    pub fn beat(&mut self) {
        if self.locked {
            return;
        }
        self.log.push(Entry::Beat { n: 1 });
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn locked_beat_logs_nothing() {
        let mut p = Pulse {
            log: Vec::new(),
            locked: true,
        };
        p.beat();
        assert_eq!(p.log, vec![]);
    }
}
