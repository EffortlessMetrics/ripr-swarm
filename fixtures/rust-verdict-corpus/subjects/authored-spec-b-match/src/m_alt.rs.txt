//! An alternative pattern in the helper's match.

pub fn label_alt(kind: &str) -> &'static str {
    match kind {
        "word" | "term" => "alpha",
        "text" => "beta",
        _ => "other",
    }
}

pub fn classify_alt(input: &str) -> &'static str {
    let final_label = label_alt(input);
    if final_label == "alpha" {
        "word"
    } else {
        "blank"
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn each_arm_has_its_own_label_m_alt() {
        assert_eq!(classify_alt("word"), "word");
        assert_eq!(classify_alt("text"), "blank");
    }
}
