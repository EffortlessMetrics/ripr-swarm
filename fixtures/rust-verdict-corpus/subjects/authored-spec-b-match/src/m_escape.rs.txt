//! A pattern literal that carries an escape.

pub fn label_escape(kind: &str) -> &'static str {
    match kind {
        "a\tb" => "alpha",
        "text" => "beta",
        _ => "other",
    }
}

pub fn classify_escape(input: &str) -> &'static str {
    let final_label = label_escape(input);
    if final_label == "alpha" {
        "word"
    } else {
        "blank"
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn each_arm_has_its_own_label_m_escape() {
        assert_eq!(classify_escape("a\tb"), "word");
        assert_eq!(classify_escape("text"), "blank");
    }
}
