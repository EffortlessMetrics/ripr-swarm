pub struct Pulse {
    pub log: Vec<u32>,
}

impl Pulse {
    pub fn beat(&mut self) {
        self.log.push(1);
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn seeded_beat_appends() {
        let mut p = Pulse { log: vec![9] };
        p.beat();
        assert_eq!(p.log, vec![9, 1]);
    }
}
