//! Authored subject for the log-capture limit: a record is bound to a
//! test only through its own format template (RIPR-SPEC-0231, #7133).

/// Logs two records; its test captures and checks only the first.
pub fn settle(amount: u32) {
    log::info!("settled {}", amount * 3);
    log::info!(target: "settle", "closing {}", amount + 1);
}

/// Logs one targeted record; its test captures and checks it.
pub fn open(amount: u32) {
    log::info!(target: "open", "opened {}", amount * 2);
}

#[cfg(test)]
mod tests {
    use super::*;
    use std::sync::{Mutex, OnceLock};

    struct Capture(Mutex<Vec<String>>);

    impl log::Log for Capture {
        fn enabled(&self, _: &log::Metadata) -> bool {
            true
        }
        fn log(&self, record: &log::Record) {
            if let Ok(mut lines) = self.0.lock() {
                lines.push(record.args().to_string());
            }
        }
        fn flush(&self) {}
    }

    fn capture() -> &'static Capture {
        static LOGGER: OnceLock<&'static Capture> = OnceLock::new();
        LOGGER.get_or_init(|| {
            let logger: &'static Capture = Box::leak(Box::new(Capture(Mutex::new(Vec::new()))));
            let _ = log::set_logger(logger);
            log::set_max_level(log::LevelFilter::Info);
            logger
        })
    }

    #[test]
    fn settle_logs_the_tripled_amount() {
        let logger = capture();
        settle(3);
        let lines = logger.0.lock().map(|l| l.clone()).unwrap_or_default();
        assert!(lines.iter().any(|l| l == "settled 9"), "{lines:?}");
    }

    #[test]
    fn open_logs_the_doubled_amount() {
        let logger = capture();
        open(4);
        let lines = logger.0.lock().map(|l| l.clone()).unwrap_or_default();
        assert!(lines.iter().any(|l| l == "opened 8"), "{lines:?}");
    }

    #[test]
    fn quote_logs_the_raw_record() {
        let logger = capture();
        quote(5);
        let lines = logger.0.lock().map(|l| l.clone()).unwrap_or_default();
        assert!(lines.iter().any(|l| l == r#"said "quoted 10" twice"#), "{lines:?}");
    }
}

/// Logs a record with embedded quotes; its test asserts it with a raw literal.
pub fn quote(amount: u32) {
    log::info!("said \"quoted {}\" twice", amount + 5);
}
