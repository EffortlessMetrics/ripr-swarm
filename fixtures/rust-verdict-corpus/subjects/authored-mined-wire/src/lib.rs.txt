//! A small wire codec whose tests use shapes mined from real crates: an
//! example in the docs as the only test, a test-generating macro that asserts
//! inside a closure, a property function returning bool, a round trip, and a
//! digit lookup table.

/// Reads a big-endian `u16` from the first two bytes.
///
/// ```
/// assert_eq!(wire::read_u16(&[0x12, 0x34]), 0x1234);
/// ```
pub fn read_u16(bytes: &[u8]) -> u16 {
    u16::from_be_bytes([bytes[0], bytes[1]])
}

/// Reads a little-endian `u16` from the first two bytes.
///
/// ```ignore
/// assert_eq!(wire::read_u16_le(&[0x34, 0x12]), 0x1234);
/// ```
pub fn read_u16_le(bytes: &[u8]) -> u16 {
    u16::from_le_bytes([bytes[0], bytes[1]])
}

/// Reads a big-endian `u32` from the first four bytes.
pub fn read_u32(bytes: &[u8]) -> u32 {
    u32::from_be_bytes([bytes[0], bytes[1], bytes[2], bytes[3]])
}

/// How far a frame header has been read.
#[derive(Debug, PartialEq, Eq)]
pub enum Status {
    Complete(usize),
    Partial,
}

/// Parses a frame header: a version byte, then a length byte.
pub fn header(bytes: &[u8]) -> Status {
    match bytes {
        [1, len, ..] => Status::Complete(*len as usize + 2),
        [2, ..] => Status::Complete(1),
        _ => Status::Partial,
    }
}

const MASK: u8 = 0x5a;

/// Masks a payload in place; masking twice restores it.
pub fn mask(payload: &mut [u8]) {
    for byte in payload.iter_mut() {
        *byte ^= MASK;
    }
}

static DIGITS: [u8; 16] = [
    b'0', b'1', b'2', b'3', b'4', b'5', b'6', b'7',
    b'8', b'9',
    b'a', b'b', b'c', b'd', b'e', b'f',
];

/// Lower-case hex for a byte slice.
pub fn to_hex(bytes: &[u8]) -> String {
    let mut out = String::with_capacity(bytes.len() * 2);
    for byte in bytes {
        out.push(DIGITS[(byte >> 4) as usize] as char);
        out.push(DIGITS[(byte & 0x0f) as usize] as char);
    }
    out
}

#[cfg(test)]
mod tests {
    use super::*;

    macro_rules! header_case {
        ($name:ident, $input:expr, |$status:ident| $body:block) => {
            #[test]
            fn $name() {
                let check = |$status: Status| $body;
                check(header($input));
            }
        };
    }

    header_case!(version_one_counts_its_length, &[1, 3, 0, 0, 0], |status| {
        assert_eq!(status, Status::Complete(5));
    });

    header_case!(version_two_is_one_byte, &[2], |status| {
        assert_eq!(status, Status::Complete(1));
    });

    header_case!(empty_is_partial, &[], |status| {
        assert_eq!(status, Status::Partial);
    });

    fn reads_back(value: u32) -> bool {
        read_u32(&value.to_be_bytes()) == value
    }

    #[test]
    fn u32_reads_what_was_written() {
        for value in [1, 0x0102_0304, 0xdead_beef, u32::MAX - 1] {
            assert!(reads_back(value));
        }
    }

    #[test]
    fn masking_twice_restores_the_payload() {
        let mut payload = *b"frame";
        mask(&mut payload);
        mask(&mut payload);
        assert_eq!(&payload, b"frame");
    }

    #[test]
    fn hex_of_decimal_digits() {
        assert_eq!(to_hex(&[0x01, 0x23, 0x45, 0x67, 0x89]), "0123456789");
    }
}
