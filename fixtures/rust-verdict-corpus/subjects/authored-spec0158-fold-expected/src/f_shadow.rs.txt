pub fn tier_shadow(qty: u32) -> u32 {
    if 10 <= qty {
        5
    } else {
        0
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn a_shadowed_base_is_fourteen_f_shadow() {
        let base = 9;
        let base = base + 5;
        assert_eq!(tier_shadow(base + 1), 5);
        assert_eq!(tier_shadow(base), 5);
    }
}
