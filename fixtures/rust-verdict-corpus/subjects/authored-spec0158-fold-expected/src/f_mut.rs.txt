pub fn tier_mut(qty: u32) -> u32 {
    if 10 <= qty {
        5
    } else {
        0
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn a_grown_base_is_fourteen_f_mut() {
        let mut base = 9;
        base += 5;
        assert_eq!(tier_mut(base + 1), 5);
        assert_eq!(tier_mut(base), 5);
    }
}
