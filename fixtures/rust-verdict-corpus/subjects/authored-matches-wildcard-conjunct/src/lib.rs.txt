#[derive(Debug, PartialEq)]
pub enum E {
    Bad,
    Big,
}

pub fn check(x: u32) -> Result<u32, E> {
    if x > 10 {
        return Err(E::Big);
    }
    Ok(x)
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn check_accepts_five() {
        let ready = true;
        assert!(ready && matches!(check(5), Ok(_)));
    }
}
