pub fn tier_hcomp(qty: u32) -> u32 {
    if 10 <= qty {
        5
    } else {
        0
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    fn extra() -> u32 {
        5
    }

    fn big() -> u32 {
        10 + extra()
    }

    fn small() -> u32 {
        9 + extra()
    }

    #[test]
    fn the_computing_helpers_are_fifteen_and_fourteen_m_hcomp() {
        assert_eq!(tier_hcomp(big()), 5);
        assert_eq!(tier_hcomp(small()), 5);
    }
}
