pub fn tier_gsib(qty: u32) -> u32 {
    if 10 <= qty {
        5
    } else {
        0
    }
}

#[allow(dead_code)]
mod hidden {
    pub fn big() -> u32 {
        10
    }

    pub fn small() -> u32 {
        9
    }
}

#[cfg(test)]
mod tests {
    use crate::m_gext::*;

    use super::*;

    #[test]
    fn the_glob_helpers_are_fifteen_and_fourteen_m_gsib() {
        assert_eq!(tier_gsib(big()), 5);
        assert_eq!(tier_gsib(small()), 5);
    }
}
