pub fn clamp_low(x: u32) -> u32 {
    if x < 5 { 5 } else { x }
}

    ((((((((((a + 1) * 2) - 1) + 3) * 2) - 2) + 4) * 2) - 3) + 5) * 2

macro_rules! def_add {
    ($name:ident, $n:expr) => {
        pub fn $name(x: u32) -> u32 {
            x + $n
        }
    };
}

def_add!(add_five, 5);

pub fn attribute_noise(x: u32) -> u32 {
    x + 7
}

pub fn commented_tail(x: u32) -> u32 {
    x + 1 // assert_eq!(commented_tail(1), 3);
}

pub fn minified(x: u32) -> u32 { x + 1 }

pub fn inc(x: u32) -> u32 {
    x + 1
}

pub fn call_through_pointer(x: u32) -> u32 {
    x * 3
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn string_mentions_the_owner() {
        let doc = "assert_eq!(quoted_only(2), 4)";
        assert!(!doc.is_empty());
    }

    #[test]
    fn raw_identifier_owner() {
        assert_eq!(r#match(1), 2);
    }

    #[test]
    fn unicode_identifier_owner() {
        assert_eq!(größe(4), 8);
    }

    #[test]
    fn long_line_owner() {
        assert_eq!(long_sum(0), 3240);
    }

    #[test]
    fn nested_owner() {
        assert_eq!(nested(1), 60);
    }

    #[test]
    fn macro_defined_owner() {
        assert_eq!(add_five(1), 6);
    }

    #[test]
    #[rustfmt::skip]
    #[allow(clippy::eq_op)]
    #[doc = "attribute noise above the test fn"]
    fn attribute_noise_owner() {
        assert_eq!(attribute_noise(1), 8);
    }

    #[cfg_attr(all(), test)]
    fn cfg_attr_test_owner() {
        assert_eq!(commented_tail(1), 2);
    }

    #[test] fn minified_owner() { assert_eq!(minified(1), 2); }

    #[test]
    fn owner_as_a_path_value() {
        assert_eq!(vec![1, 2].into_iter().map(inc).collect::<Vec<_>>(), vec![2, 3]);
    }

    #[test]
    fn owner_through_a_fn_pointer() {
        let f: fn(u32) -> u32 = call_through_pointer;
        assert_eq!(f(2), 6);
    }
}
