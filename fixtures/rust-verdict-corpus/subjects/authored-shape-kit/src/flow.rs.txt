use std::cell::RefCell;

pub fn word_len_untested_else(s: &str) -> usize {
    let Some(w) = s.split_whitespace().next() else {
        return 0;
    };
    w.len()
}

pub fn word_len_tested_else(s: &str) -> usize {
    let Some(w) = s.split_whitespace().next() else {
        return 0;
    };
    w.len()
}

pub fn find_cell(grid: &[Vec<u32>], target: u32) -> Option<(usize, usize)> {
    let mut found = None;
    'outer: for (i, row) in grid.iter().enumerate() {
        for (j, &v) in row.iter().enumerate() {
            if v == target {
                found = Some((i, j));
                break 'outer;
            }
        }
    }
    found
}

pub fn ends(xs: &[u32]) -> Option<(u32, u32)> {
    match xs {
        [] => None,
        [x] => Some((*x, *x)),
        [first, .., last] => Some((*first, *last)),
    }
}

pub fn low_byte(n: u32) -> u8 {
    n as u8
}

pub fn bump(x: u8) -> u8 {
    x.saturating_add(10)
}

pub fn parse_all(xs: &[&str]) -> Result<Vec<u32>, std::num::ParseIntError> {
    xs.iter().map(|s| s.parse::<u32>()).collect()
}

pub fn name_len(n: Option<&str>) -> usize {
    n.map_or(0, |s| s.len())
}

pub fn top_two(mut v: Vec<u32>) -> Vec<u32> {
    v.sort_by(|a, b| b.cmp(a));
    v.truncate(2);
    v
}

pub fn rising(xs: &[u32]) -> usize {
    xs.windows(2).filter(|w| w[1] > w[0]).count()
}

pub struct Tally {
    hits: RefCell<u32>,
}

impl Tally {
    pub fn new() -> Tally {
        Tally { hits: RefCell::new(0) }
    }

    pub fn hit(&self) {
        *self.hits.borrow_mut() += 1;
    }

    pub fn hits(&self) -> u32 {
        *self.hits.borrow()
    }
}

pub fn doubled_on_thread(x: u32) -> u32 {
    std::thread::spawn(move || x * 2).join().unwrap()
}

pub fn make_adder(n: i32) -> impl Fn(i32) -> i32 {
    move |x| x + n
}

pub fn make_offset(n: i32) -> impl Fn(i32) -> i32 {
    move |x| x + n
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn word_len_of_a_sentence() {
        assert_eq!(word_len_untested_else("hello world"), 5);
    }

    #[test]
    fn word_len_of_blank_text() {
        assert_eq!(word_len_tested_else("   "), 0);
        assert_eq!(word_len_tested_else("ab cd"), 2);
    }

    #[test]
    fn find_cell_in_second_row() {
        let grid = vec![vec![1, 2], vec![3, 4]];
        assert_eq!(find_cell(&grid, 3), Some((1, 0)));
    }

    #[test]
    fn ends_of_longer_slices() {
        assert_eq!(ends(&[1, 2, 3]), Some((1, 3)));
        assert_eq!(ends(&[]), None);
    }

    #[test]
    fn low_byte_of_a_word() {
        assert_eq!(low_byte(0x1234), 0x34);
    }

    #[test]
    fn bump_saturates() {
        assert_eq!(bump(250), 255);
    }

    #[test]
    fn parse_all_good_and_bad() {
        assert_eq!(parse_all(&["1", "2"]), Ok(vec![1, 2]));
        assert!(parse_all(&["1", "x"]).is_err());
    }

    #[test]
    fn name_len_of_some() {
        assert_eq!(name_len(Some("abc")), 3);
    }

    #[test]
    fn top_two_descending() {
        assert_eq!(top_two(vec![3, 9, 5]), vec![9, 5]);
    }

    #[test]
    fn rising_pairs() {
        assert_eq!(rising(&[1, 3, 2, 4]), 2);
    }

    #[test]
    fn tally_counts_hits() {
        let t = Tally::new();
        t.hit();
        t.hit();
        assert_eq!(t.hits(), 2);
    }

    #[test]
    fn thread_doubles() {
        assert_eq!(doubled_on_thread(5), 10);
    }

    #[test]
    fn adder_adds() {
        assert_eq!(make_adder(2)(3), 5);
    }

    #[test]
    fn offset_of_zero() {
        assert_eq!(make_offset(0)(7), 7);
    }
}
