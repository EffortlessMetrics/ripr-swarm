use std::cell::Cell;
use std::cmp::Ordering;
use std::ops::{Add, Neg};
use std::rc::Rc;
use std::str::FromStr;

#[derive(Debug, Clone, Copy, PartialEq)]
pub struct Money(pub i64);

impl Add for Money {
    type Output = Money;
    fn add(self, other: Money) -> Money {
        Money(self.0 + other.0)
    }
}

impl Neg for Money {
    type Output = Money;
    fn neg(self) -> Money {
        Money(-self.0)
    }
}

#[derive(Debug, PartialEq)]
pub struct Celsius(pub i32);

#[derive(Debug, PartialEq)]
pub struct Fahrenheit(pub i32);

impl From<Celsius> for Fahrenheit {
    fn from(c: Celsius) -> Fahrenheit {
        Fahrenheit(c.0 * 9 / 5 + 32)
    }
}

#[derive(Debug, PartialEq)]
pub struct Port(pub u16);

#[derive(Debug, PartialEq)]
pub enum PortError {
    NotANumber,
    TooBig,
}

impl FromStr for Port {
    type Err = PortError;
    fn from_str(s: &str) -> Result<Port, PortError> {
        let n: u32 = s.parse().map_err(|_| PortError::NotANumber)?;
        if n > 65535 {
            return Err(PortError::TooBig);
        }
        Ok(Port(n as u16))
    }
}

pub struct Countdown(pub u32);

impl Iterator for Countdown {
    type Item = u32;
    fn next(&mut self) -> Option<u32> {
        if self.0 == 0 {
            return None;
        }
        self.0 -= 1;
        Some(self.0 + 1)
    }
}

#[derive(Debug, PartialEq)]
pub struct Version(pub u32, pub u32);

impl PartialOrd for Version {
    fn partial_cmp(&self, other: &Version) -> Option<Ordering> {
        (self.0, self.1).partial_cmp(&(other.0, other.1))
    }
}

#[derive(Debug, PartialEq)]
pub struct Settings {
    pub retries: u32,
    pub verbose: bool,
}

impl Default for Settings {
    fn default() -> Settings {
        Settings { retries: 3, verbose: false }
    }
}

pub struct Guard(pub Rc<Cell<u32>>);

impl Drop for Guard {
    fn drop(&mut self) {
        self.0.set(self.0.get() + 1);
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn money_adds_through_the_operator() {
        assert_eq!(Money(2) + Money(3), Money(5));
    }

    #[test]
    fn negation_is_only_exercised() {
        let _ = -Money(1);
    }

    #[test]
    fn boiling_point_converts() {
        assert_eq!(Fahrenheit::from(Celsius(100)), Fahrenheit(212));
    }

    #[test]
    fn small_ports_parse() {
        assert_eq!("80".parse::<Port>(), Ok(Port(80)));
        assert_eq!("70000".parse::<Port>(), Err(PortError::TooBig));
        assert_eq!("http".parse::<Port>(), Err(PortError::NotANumber));
    }

    #[test]
    fn countdown_yields_descending() {
        assert_eq!(Countdown(3).collect::<Vec<_>>(), vec![3, 2, 1]);
    }

    #[test]
    fn minor_versions_order() {
        assert!(Version(1, 2) < Version(1, 3));
    }

    #[test]
    fn retries_default() {
        assert_eq!(Settings::default().retries, 3);
    }

    #[test]
    fn guard_counts_its_drop() {
        let counter = Rc::new(Cell::new(0));
        {
            let _g = Guard(counter.clone());
        }
        assert_eq!(counter.get(), 1);
    }
}
