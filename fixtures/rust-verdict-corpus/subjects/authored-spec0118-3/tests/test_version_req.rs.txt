use spec0118_3::{Version, VersionReq};

fn req(s: &str) -> VersionReq {
    VersionReq::parse(s).expect("a valid requirement")
}

fn assert_match_all(req: &VersionReq, versions: &[(u64, u64)]) {
    for &(major, minor) in versions {
        assert!(req.matches(&Version { major, minor }), "{major}.{minor} should match");
    }
}

fn assert_match_none(req: &VersionReq, versions: &[(u64, u64)]) {
    for &(major, minor) in versions {
        assert!(!req.matches(&Version { major, minor }), "{major}.{minor} should not match");
    }
}

#[test]
fn test_greater() {
    let r = req(">1.2");
    assert_match_all(&r, &[(1, 3), (2, 0)]);
    assert_match_none(&r, &[(1, 2), (1, 1), (0, 9)]);
}
