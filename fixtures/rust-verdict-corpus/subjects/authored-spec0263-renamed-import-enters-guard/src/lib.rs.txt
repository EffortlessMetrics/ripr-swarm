/// Records `x` only when it is large.
pub fn add(v: &mut Vec<u32>, x: u32) {
    if x > 10 {
        v.push(x);
    }
}

#[cfg(test)]
mod tests {
    use super::add;
    use super::add as push_big;

    #[test]
    fn records_through_the_alias() {
        let mut v = Vec::new();
        add(&mut v, 4);
        push_big(&mut v, 11);
        assert_eq!(v, vec![11]);
    }
}
