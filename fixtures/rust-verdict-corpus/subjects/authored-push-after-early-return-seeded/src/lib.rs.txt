#[derive(Debug, PartialEq)]
pub enum Entry {
    Beat { n: u32 },
}

pub struct Pulse {
    pub log: Vec<Entry>,
    pub locked: bool,
}

impl Pulse {
    pub fn beat(&mut self) {
        if self.locked {
            return;
        }
        self.log.push(Entry::Beat { n: 1 });
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn locked_beat_keeps_seeded_log() {
        let mut p = Pulse {
            log: vec![Entry::Beat { n: 9 }],
            locked: true,
        };
        p.beat();
        assert_eq!(p.log, vec![Entry::Beat { n: 9 }]);
    }
}
