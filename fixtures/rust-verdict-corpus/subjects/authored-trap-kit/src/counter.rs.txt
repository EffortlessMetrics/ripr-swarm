#[derive(Clone, Debug, Default)]
pub struct Card {
    pub points: u32,
}

pub fn with_bonus(base: &Card) -> Card {
    Card {
        points: base.points * 3,
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn bonus_triples_the_points() {
        let card = Card { points: 2 };
        let snapshot = card.clone();
        let next = with_bonus(&card);
        assert_eq!(snapshot.points, 2);
        let _ = next;
    }
}
