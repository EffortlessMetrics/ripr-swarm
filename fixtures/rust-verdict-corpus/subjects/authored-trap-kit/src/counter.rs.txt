#[derive(Clone, Debug, Default)]
pub struct Counter {
    pub n: u32,
}

pub fn with_bonus(base: &Counter) -> Counter {
    Counter {
        n: base.n * 3,
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn bonus_moves_the_counter() {
        let counter = Counter { n: 2 };
        let snapshot = counter.clone();
        let next = with_bonus(&counter);
        assert_eq!(snapshot.n, 2);
        let _ = next;
    }
}
