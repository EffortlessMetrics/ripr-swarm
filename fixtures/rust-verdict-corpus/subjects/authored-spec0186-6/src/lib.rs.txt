//! A private predicate reached only through a public wrapper whose test pins the edge.

fn is_bulk(qty: u32) -> bool { qty >= 10 }

pub fn order_discount(qty: u32) -> u32 {
    if is_bulk(qty) { 5 } else { 0 }
}

#[cfg(test)]
mod tests {
    use super::order_discount;

    #[test]
    fn bulk_edge_is_discounted() {
        assert_eq!(order_discount(10), 5);
        assert_eq!(order_discount(9), 0);
    }
}
