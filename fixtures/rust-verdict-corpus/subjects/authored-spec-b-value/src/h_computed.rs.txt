//! The call site passes a computed argument; the reach relation holds but the
//! row value does not transfer.

fn is_wide(input: &str) -> bool {
    input.len() > 3
}

pub fn classify_trimmed(input: &str) -> &'static str {
    if is_wide(input.trim()) {
        "wide"
    } else {
        "narrow"
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn rows_straddle_four_computed() {
        assert_eq!(classify_trimmed(" abcd "), "wide");
        assert_eq!(classify_trimmed(" abc "), "narrow");
    }
}
