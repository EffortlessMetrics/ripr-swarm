//! The chain goes through a method outside the evaluated families.

pub fn upper_boundary_hit(input: &str, delim: char) -> bool {
    let end = input.to_ascii_uppercase().rfind(delim).map_or(1, |idx| idx);
    let start = delim.len_utf8();
    end == start
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn absent_delimiter_lands_on_the_boundary_v_method() {
        assert!(upper_boundary_hit("ab", 'X'));
    }

    #[test]
    fn present_delimiter_misses_the_boundary_v_method() {
        assert!(!upper_boundary_hit("Xab", 'X'));
        assert!(!upper_boundary_hit("abx", 'X'));
    }
}
