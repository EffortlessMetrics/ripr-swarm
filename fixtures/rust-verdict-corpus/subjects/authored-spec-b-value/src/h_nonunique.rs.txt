//! The helper's name is not unique in the workspace.

fn wide_nu(input: &str) -> bool {
    input.len() > 3
}

pub mod other {
    pub fn wide_nu(input: &str) -> bool {
        input.len() > 4
    }
}

pub fn classify_ambiguous(input: &str) -> &'static str {
    if wide_nu(input) {
        "wide"
    } else {
        "narrow"
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn rows_straddle_four_nonunique_h_nonunique() {
        assert_eq!(classify_ambiguous("abcd"), "wide");
        assert_eq!(classify_ambiguous("abc"), "narrow");
    }

    #[test]
    fn the_other_module_needs_five_bytes_h_nonunique() {
        assert!(other::wide_nu("abcde"));
        assert!(!other::wide_nu("abcd"));
    }
}
