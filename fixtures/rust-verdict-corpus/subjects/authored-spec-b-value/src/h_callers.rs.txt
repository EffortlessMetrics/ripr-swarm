//! The helper has two callers.

fn is_wide(input: &str) -> bool {
    input.len() > 3
}

pub fn classify_shared(input: &str) -> &'static str {
    if is_wide(input) {
        "wide"
    } else {
        "narrow"
    }
}

pub fn label_shared(input: &str) -> String {
    format!("{}:{}", input, is_wide(input))
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn rows_straddle_four_callers() {
        assert_eq!(classify_shared("abcd"), "wide");
        assert_eq!(classify_shared("abc"), "narrow");
    }

    #[test]
    fn the_label_carries_the_flag() {
        assert_eq!(label_shared("abcde"), "abcde:true");
    }
}
