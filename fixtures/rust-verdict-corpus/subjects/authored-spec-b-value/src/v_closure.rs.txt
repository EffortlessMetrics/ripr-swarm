//! The map_or closure is not the identity.

pub fn closure_boundary_hit(input: &str, delim: char) -> bool {
    let end = input.rfind(delim).map_or(1, |idx| idx + 0);
    let start = delim.len_utf8();
    end == start
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn absent_delimiter_lands_on_the_boundary_v_closure() {
        assert!(closure_boundary_hit("ab", 'x'));
    }

    #[test]
    fn present_delimiter_misses_the_boundary_v_closure() {
        assert!(!closure_boundary_hit("xab", 'x'));
        assert!(!closure_boundary_hit("abx", 'x'));
    }
}
