//! The helper calls itself.

fn is_wide(input: &str, depth: usize) -> bool {
    if depth == 0 {
        input.len() > 3
    } else {
        is_wide(input, depth - 1)
    }
}

pub fn classify_recursive(input: &str) -> &'static str {
    if is_wide(input, 1) {
        "wide"
    } else {
        "narrow"
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn rows_straddle_four_recursion() {
        assert_eq!(classify_recursive("abcd"), "wide");
        assert_eq!(classify_recursive("abc"), "narrow");
    }
}
