//! The helper is reached through a path-qualified call site.

mod meter {
    pub fn is_wide(input: &str) -> bool {
        input.len() > 3
    }
}

pub fn classify_path(input: &str) -> &'static str {
    if meter::is_wide(input) {
        "wide"
    } else {
        "narrow"
    }
}

#[cfg(test)]
mod tests {
    use super::*;

    #[test]
    fn rows_straddle_four_path() {
        assert_eq!(classify_path("abcd"), "wide");
        assert_eq!(classify_path("abc"), "narrow");
    }
}
